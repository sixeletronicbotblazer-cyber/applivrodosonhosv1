   199→function ReviewTab() {
   200→  const [items, setItems] = useState<AdminEntry[] | null>(null)
   201→  const [total, setTotal] = useState(0)
   202→  const [page, setPage] = useState(1)
   203→  const [status, setStatus] = useState<'all' | 'draft' | 'published'>('draft')
   204→  const [source, setSource] = useState<'all' | 'pdf_literal' | 'editorial'>('all')
   205→  const [q, setQ] = useState('')
   206→  const [selected, setSelected] = useState<Set<string>>(new Set())
   207→  const [editing, setEditing] = useState<AdminEntry | null>(null)
   208→  const [pageImages, setPageImages] = useState<Map<number, string>>(new Map())
   209→  const [busy, setBusy] = useState(false)
   210→  const [reloadKey, setReloadKey] = useState(0)
   211→  const reload = useCallback(() => setReloadKey((k) => k + 1), [])
   212→
   213→  useEffect(() => {
   214→    let alive = true
   215→    api<{ items: AdminEntry[]; total: number }>(`/api/admin/entries?status=${status}&source=${source}&q=${encodeURIComponent(q)}&page=${page}`)
   216→      .then((res) => {
   217→        if (!alive) return
   218→        setItems(res.items)
   219→        setTotal(res.total)
   220→      })
   221→      .catch(() => undefined)
   222→    return () => {
   223→      alive = false
   224→    }
   225→  }, [status, source, q, page, reloadKey])
   226→
   227→  useEffect(() => {
   228→    api<{ pages: { pdfPage: number; imagePath: string | null }[] }>('/api/book/pages')
   229→      .then((res) => {
   230→        const m = new Map<number, string>()
   231→        for (const p of res.pages) if (p.imagePath) m.set(p.pdfPage, p.imagePath)
   232→        setPageImages(m)
   233→      })
   234→      .catch(() => undefined)
   235→  }, [])
   236→
   237→  async function bulk(action: 'publish' | 'unpublish') {
   238→    setBusy(true)
   239→    try {
   240→      const res = await api<{ published: number; failed: { id: string; reason: string }[] }>('/api/admin/entries/bulk', {
   241→        method: 'POST',
   242→        body: JSON.stringify({ ids: [...selected], action }),
   243→      })
   244→      toast({ title: `${res.published} verbetes processados`, description: res.failed.length ? `${res.failed.length} falharam (ver regra de publicação).` : undefined })
   245→      setSelected(new Set())
   246→      reload()
   247→    } catch (e) {
   248→      toast({ title: e instanceof ApiError ? e.message : 'Falha na operação' })
   249→    } finally {
   250→      setBusy(false)
   251→    }
   252→  }
   253→
   254→  const totalPages = Math.max(1, Math.ceil(total / 20))
   255→
   256→  return (
   257→    <div>
   258→      <div className="flex flex-wrap items-center gap-2">
   259→        <select value={status} onChange={(e) => { setStatus(e.target.value as typeof status); setPage(1) }} aria-label="Filtrar por status" className="focus-ring h-10 rounded-xl border border-linha bg-white px-3 text-sm">
   260→          <option value="draft">Em curadoria</option>
   261→          <option value="published">Publicados</option>
   262→          <option value="all">Todos</option>
   263→        </select>
   264→        <select value={source} onChange={(e) => { setSource(e.target.value as typeof source); setPage(1) }} aria-label="Filtrar por origem" className="focus-ring h-10 rounded-xl border border-linha bg-white px-3 text-sm">
   265→          <option value="all">PDF + editoriais</option>
   266→          <option value="pdf_literal">Só do PDF</option>
   267→          <option value="editorial">Só editoriais</option>
   268→        </select>
   269→        <Input value={q} onChange={(e) => { setQ(e.target.value); setPage(1) }} placeholder="Buscar título…" aria-label="Buscar título" className="h-10 w-40 rounded-xl border-linha bg-white text-sm" />
   270→        <span className="ml-auto text-xs text-muted-foreground">{total} verbetes</span>
   271→      </div>
   272→
   273→      {selected.size > 0 ? (
   274→        <div className="sticky top-2 z-10 mt-3 flex flex-wrap items-center gap-2 rounded-xl bg-roxo px-4 py-2.5 text-white">
   275→          <span className="text-sm font-semibold">{selected.size} selecionados</span>
   276→          <Button onClick={() => void bulk('publish')} disabled={busy} className="focus-ring h-9 rounded-lg bg-ouro px-3 text-xs font-bold text-roxo hover:bg-[#EBBB3E]">
   277→            Publicar selecionados
   278→          </Button>
   279→          <Button onClick={() => void bulk('unpublish')} disabled={busy} variant="outline" className="focus-ring h-9 rounded-lg border-white/30 bg-transparent px-3 text-xs font-bold text-white hover:bg-white/10">
   280→            Devolver para draft
   281→          </Button>
   282→          <button type="button" onClick={() => setSelected(new Set())} className="focus-ring ml-auto text-xs underline">
   283→            limpar
   284→          </button>
   285→        </div>
   286→      ) : null}
   287→
   288→      <div className="mt-4 space-y-2.5">
   289→        {!items ? (
   290→          <Skeleton className="h-64 w-full rounded-2xl bg-lilas/50" />
   291→        ) : items.length === 0 ? (
   292→          <EmptyBox text="Nenhum verbete com esses filtros." />
   293→        ) : (
   294→          items.map((e) => <ReviewRow key={e.id} entry={e} pageImage={pageImages.get(e.pdfPage ?? -1) ?? null} selected={selected.has(e.id)} onToggle={() => {
   295→            const next = new Set(selected)
   296→            if (next.has(e.id)) next.delete(e.id)
   297→            else next.add(e.id)
   298→            setSelected(next)
   299→          }} onEdit={() => setEditing(e)} onChanged={reload} />)
   300→        )}
   301→      </div>
   302→
   303→      {totalPages > 1 ? (
   304→        <div className="mt-4 flex items-center justify-center gap-3">
   305→          <Button variant="outline" disabled={page <= 1} onClick={() => setPage((p) => p - 1)} className="focus-ring h-10 rounded-xl border-linha bg-white px-4 text-sm text-roxo">
   306→            ← Anterior
   307→          </Button>
   308→          <span className="text-sm text-muted-foreground">
   309→            {page} / {totalPages}
   310→          </span>
   311→          <Button variant="outline" disabled={page >= totalPages} onClick={() => setPage((p) => p + 1)} className="focus-ring h-10 rounded-xl border-linha bg-white px-4 text-sm text-roxo">
   312→            Próxima →
   313→          </Button>
   314→        </div>
   315→      ) : null}
   316→
   317→      {editing ? <EditDialog entry={editing} pageImage={pageImages.get(editing.pdfPage ?? -1) ?? null} onClose={() => setEditing(null)} onChanged={reload} /> : null}
   318→    </div>
   319→  )
   320→}
